\chapter{Group Character Expansions and Plaquette Expectations for \texorpdfstring{$\SU(N)$}{SU(N)}}
\label{app:characters}

\section{Introduction and Mathematical Scope}
\label{sec:app_a_intro}

In this appendix, we establish the exact group-theoretic foundations of character expansions for the non-Abelian gauge groups $G = \SU(N)$ ($N \ge 2$). Group character expansions constitute one of the most powerful analytical techniques in non-perturbative lattice gauge theory: by decomposing the non-linear Boltzmann weight $\exp(-S_W(U))$ into an orthogonal series over irreducible representation characters, the non-local multi-link functional integral is transformed into an exact discrete sum over invariant tensor networks.

These expansions provide the computational and analytical backbone for:
\begin{enumerate}
    \item The single-cell contractivity bounds $\zeta_R(\beta) < 1$ utilised in the Fr\"ohlich--Lieb--Seiler chessboard estimates (Chapter~\ref{ch:positivity_chessboards} and Appendix~\ref{app:chessboard_estimates});
    \item The high-temperature polymer cluster expansion and Koteck\'y--Preiss convergence across the crossover window $\mathcal{W} = [1/2, 5/2]$ (Chapter~\ref{ch:crossover_closure_dsca});
    \item The exact strong-coupling area law for planar Wilson loops, yielding the string tension bound $\sigma_R(\beta) \ge -\ln \zeta_R(\beta) > 0$.
\end{enumerate}

\section{Haar Measure, Peter--Weyl Orthogonality, and Weyl Integration}
\label{sec:haar_peter_weyl}

\subsection{Haar Measure and the Peter--Weyl Theorem}
\label{subsec:peter_weyl_ortho_app}

Let $G$ be a compact, connected Lie group endowed with its unique normalized, bi-invariant Haar measure $dU$, satisfying:
\begin{equation}
\int_G dU = 1, \qquad \int_G d(g U) = \int_G d(U g) = \int_G dU \quad \forall \, g \in G.
\end{equation}
Let $\widehat{G}$ denote the set of equivalence classes of finite-dimensional irreducible complex representations of $G$. For each equivalence class $R \in \widehat{G}$, let $d_R \coloneqq \dim(R)$ denote its representation dimension, and let:
\begin{equation}
\chi_R(U) \coloneqq \Tr_R(U), \qquad U \in G,
\end{equation}
denote its group character. A complex-valued function $f \in L^2(G, dU)$ is called a \emph{class function} (or central function) if it is invariant under group conjugation:
\begin{equation}
f(g U g^{-1}) = f(U) \qquad \forall \, g, U \in G.
\end{equation}
Let $L^2_{\mathrm{class}}(G, dU)$ denote the closed subspace of square-integrable class functions.

\begin{theorem}[Peter--Weyl Orthogonality and Completeness]
\label{thm:peter_weyl}
The characters $\{\chi_R\}_{R \in \widehat{G}}$ form a complete orthonormal basis of the Hilbert space of class functions $L^2_{\mathrm{class}}(G, dU)$:
\begin{equation}
\label{eq:peter_weyl_ortho}
\int_G dU \, \chi_R(U) \, \overline{\chi_{R'}(U)} = \delta_{R, R'}.
\end{equation}
Every central function $f \in L^2_{\mathrm{class}}(G)$ admits an unconditionally convergent Fourier--Laplace character expansion:
\begin{equation}
\label{eq:character_expansion_general_app}
f(U) = \sum_{R \in \widehat{G}} c_R \, d_R \, \chi_R(U),
\end{equation}
where the character Fourier coefficients are defined by the projection integrals:
\begin{equation}
\label{eq:cr_coefficient_def}
c_R \coloneqq \frac{1}{d_R} \int_G dU \, f(U) \, \overline{\chi_R(U)}.
\end{equation}
Furthermore, the $L^2$ Parseval identity holds:
\begin{equation}
\label{eq:parseval_characters}
\int_G dU \, |f(U)|^2 = \sum_{R \in \widehat{G}} d_R^2 \, |c_R|^2.
\end{equation}
\end{theorem}

\subsection{The Weyl Integration Formula for \texorpdfstring{$\SU(N)$}{SU(N)}}
\label{subsec:weyl_integration_formula}

Because class functions depend only on the eigenvalues of $U$, the $(N^2 - 1)$-dimensional group integral over $\SU(N)$ reduces to an $(N-1)$-dimensional integral over the maximal torus $T \subset \SU(N)$.

Any unitary matrix $U \in \SU(N)$ can be conjugated into diagonal form:
\begin{equation}
U \sim \mathrm{diag}\left( e^{i\theta_1}, e^{i\theta_2}, \dots, e^{i\theta_N} \right), \qquad \sum_{j=1}^N \theta_j \equiv 0 \pmod{2\pi},
\end{equation}
where $\theta_j \in [-\pi, \pi]$ are the eigenphases.

\begin{theorem}[Weyl Integration Formula]
\label{thm:weyl_integration}
Let $f \in L^1_{\mathrm{class}}(\SU(N))$ be a class function. Then:
\begin{equation}
\label{eq:weyl_integration_formula}
\int_{\SU(N)} dU \, f(U) = \frac{1}{N!} \left( \prod_{j=1}^N \int_{-\pi}^\pi \frac{d\theta_j}{2\pi} \right) 2\pi \delta\left( \sum_{k=1}^N \theta_k \right) \Delta(\theta) \overline{\Delta(\theta)} \, f(e^{i\theta_1}, \dots, e^{i\theta_N}),
\end{equation}
where $\Delta(\theta)$ is the Vandermonde determinant of the eigenvalues:
\begin{equation}
\label{eq:vandermonde_determinant}
\Delta(\theta) \coloneqq \prod_{1 \le j < k \le N} \left( e^{i\theta_j} - e^{i\theta_k} \right) = \det\left[ e^{i(N - j)\theta_k} \right]_{j, k = 1}^N.
\end{equation}
The Haar volume element on the maximal torus is the squared modulus of the Vandermonde determinant:
\begin{equation}
\label{eq:vandermonde_squared}
|\Delta(\theta)|^2 = \prod_{1 \le j < k \le N} \left| e^{i\theta_j} - e^{i\theta_k} \right|^2 = \prod_{1 \le j < k \le N} 4 \sin^2\left( \frac{\theta_j - \theta_k}{2} \right).
\end{equation}
\end{theorem}

\section{Character Expansion of the Single-Plaquette Action}
\label{sec:plaquette_character_expansion}

\subsection{Fourier--Laplace Representation of the Boltzmann Weight}
\label{subsec:boltzmann_character_expansion}

In the Wilson formulation of lattice gauge theory on a four-dimensional Euclidean lattice, the Boltzmann weight associated with a single oriented plaquette $P$ is:
\begin{equation}
\label{eq:plaquette_boltzmann_app}
w_P(U) \coloneqq \exp\left( -S_P(U) \right) = \exp\left( \frac{\beta}{2N} \left( \Tr(U) + \Tr(U^\dagger) \right) \right) = \exp\left( \frac{\beta}{N} \RePart \Tr(U) \right).
\end{equation}
Because $\Tr(g U g^{-1}) = \Tr(U)$ for all $g \in G$, $w_P$ is an analytic class function on $G$. By Theorem~\ref{thm:peter_weyl}, it expands into the convergent character series:
\begin{equation}
\label{eq:boltzmann_char_series}
w_P(U) = c_0(\beta) \left[ 1 + \sum_{R \neq 0} d_R \, \zeta_R(\beta) \, \chi_R(U) \right],
\end{equation}
where the summation runs over all non-trivial irreducible representations $R \in \widehat{G} \setminus \{0\}$, $c_0(\beta)$ is the single-plaquette partition function:
\begin{equation}
\label{eq:c0_def_app}
c_0(\beta) \coloneqq \int_G dU \exp\left( \frac{\beta}{N} \RePart \Tr(U) \right),
\end{equation}
and $\zeta_R(\beta)$ is the normalized single-cell character ratio:
\begin{equation}
\label{eq:character_ratio_def}
\zeta_R(\beta) \coloneqq \frac{c_R(\beta)}{c_0(\beta)} = \frac{1}{d_R \, c_0(\beta)} \int_G dU \, \exp\left( \frac{\beta}{N} \RePart \Tr(U) \right) \overline{\chi_R(U)}.
\end{equation}

\subsection{Strict Contractivity of Character Ratios}
\label{subsec:strict_contractivity_proof}

\begin{proposition}[Strict Contractivity Theorem for Character Ratios]
\label{prop:character_ratio_bound}
For every compact simple Lie group $G = \SU(N)$, every non-trivial irreducible representation $R \neq 0$, and every finite gauge coupling $\beta \in [0, \infty)$:
\begin{equation}
\label{eq:zeta_strict_less_one}
0 \le \zeta_R(\beta) < 1.
\end{equation}
Furthermore, as $\beta \to 0$ (strong coupling):
\begin{equation}
\label{eq:strong_coupling_zeta_scaling}
\zeta_R(\beta) = \frac{1}{|R|!} \left( \frac{\beta}{2N} \right)^{|R|} \left( 1 + \mathcal{O}(\beta^2) \right),
\end{equation}
where $|R|$ denotes the number of boxes in the Young tableau associated with representation $R$.
\end{proposition}

\begin{proof}
Consider the Hilbert space $L^2(G, dU)$ with standard inner product $\langle f, g \rangle = \int_G dU \, \overline{f(U)} g(U)$.
By definition:
\begin{equation}
c_0(\beta) = \langle 1, w_P \rangle, \qquad d_R c_R(\beta) = \langle \chi_R, w_P \rangle.
\end{equation}
Applying the Cauchy--Schwarz inequality to the functions $\chi_R$ and $w_P$:
\begin{equation}
\label{eq:cs_character_bound}
\left| \langle \chi_R, w_P \rangle \right|^2 \le \|\chi_R\|_{L^2}^2 \, \|w_P\|_{L^2}^2.
\end{equation}
By the Peter--Weyl orthogonality theorem \eqref{eq:peter_weyl_ortho}, $\|\chi_R\|_{L^2}^2 = 1$ for every irreducible representation. Thus:
\begin{equation}
d_R^2 |c_R(\beta)|^2 \le \int_G dU |w_P(U)|^2 = c_0(2\beta).
\end{equation}
To prove the strict inequality $\zeta_R(\beta) < 1$, suppose for contradiction that equality holds in the Cauchy--Schwarz inequality:
\begin{equation}
|\langle \chi_R, w_P \rangle| = \|\chi_R\|_{L^2} \, \|w_P\|_{L^2}.
\end{equation}
Equality in the Cauchy--Schwarz inequality occurs if and only if the two functions are linearly dependent:
\begin{equation}
\chi_R(U) = \lambda w_P(U) \qquad \text{for almost all } U \in G, \; \lambda \in \mathbb{C}.
\end{equation}
However, the Boltzmann weight $w_P(U) = \exp\left( \frac{\beta}{N} \RePart \Tr(U) \right)$ is strictly positive everywhere on $G$:
\begin{equation}
w_P(U) > 0 \qquad \forall \, U \in G.
\end{equation}
In contrast, for any non-trivial irreducible representation $R \neq 0$, the orthogonality to the trivial representation implies:
\begin{equation}
\int_G dU \, \chi_R(U) = \langle 1, \chi_R \rangle = 0.
\end{equation}
A strictly positive function cannot be proportional to a function of zero mean unless $\lambda = 0$, which would mean $\chi_R \equiv 0$, contradicting $\|\chi_R\|_{L^2} = 1$. 

Therefore, linear dependence is impossible for any finite $\beta < \infty$. We conclude that:
\begin{equation}
d_R c_R(\beta) < d_R c_0(\beta) \implies \zeta_R(\beta) < 1 \qquad \forall \, \beta \in [0, \infty).
\end{equation}
Non-negativity for standard real and self-conjugate representations follows from the positivity of the integrand on the maximal torus. The strong-coupling Taylor expansion \eqref{eq:strong_coupling_zeta_scaling} follows directly from expanding the exponential $\exp\left( \frac{\beta}{2N} \Tr(U) \right) = \sum_{m=0}^\infty \frac{1}{m!} \left( \frac{\beta}{2N} \right)^m (\Tr U)^m$ and applying the Schur--Weyl duality theorem.
\end{proof}

\section{Explicit Analytical Evaluation for \texorpdfstring{$\SU(2)$}{SU(2)}}
\label{sec:explicit_su2}

\subsection{Bessel Representation and Character Integrals}
\label{subsec:su2_bessel_formulas}

For the gauge group $G = \SU(2)$, every matrix $U \in \SU(2)$ is conjugate to:
\begin{equation}
U \sim \begin{pmatrix} e^{i\theta} & 0 \\ 0 & e^{-i\theta} \end{pmatrix}, \qquad \theta \in [0, \pi].
\end{equation}
The normalized Haar measure on the class angle $\theta$ is:
\begin{equation}
\label{eq:su2_haar_measure_theta}
dU = \frac{2}{\pi} \sin^2\theta \, d\theta.
\end{equation}
Irreducible representations of $\SU(2)$ are completely classified by their spin $j \in \{0, \frac{1}{2}, 1, \frac{3}{2}, \dots\}$, with dimension $d_j = 2j + 1$ and character:
\begin{equation}
\label{eq:su2_spin_j_character}
\chi_j(\theta) = \frac{\sin((2j+1)\theta)}{\sin\theta}.
\end{equation}
The single-plaquette Wilson action on $\SU(2)$ is:
\begin{equation}
\frac{\beta}{2} \Tr(U) = \beta \cos\theta.
\end{equation}

\begin{proposition}[Modified Bessel Representations for $\SU(2)$]
\label{prop:su2_bessel}
The single-plaquette partition function $c_0(\beta)$ and character coefficients $c_j(\beta)$ for $\SU(2)$ are given analytically by modified Bessel functions of the first kind $I_n(\beta)$:
\begin{align}
c_0(\beta) &= \frac{2}{\pi} \int_0^\pi d\theta \, \sin^2\theta \, e^{\beta \cos\theta} = \frac{2}{\beta} I_1(\beta), \label{eq:c0_su2} \\
c_j(\beta) &= \frac{1}{2j+1} \frac{2}{\pi} \int_0^\pi d\theta \, \sin\theta \sin((2j+1)\theta) \, e^{\beta \cos\theta} = \frac{2}{\beta (2j+1)} \left[ I_{2j}(\beta) - I_{2j+2}(\beta) \right]. \label{eq:cj_su2}
\end{align}
In particular, for the fundamental representation $j = 1/2$ ($d_{1/2} = 2$):
\begin{equation}
\label{eq:c_half_su2}
c_{1/2}(\beta) = \frac{1}{\beta} \left[ I_1(\beta) - I_3(\beta) \right] = \frac{2}{\beta^2} I_2(\beta).
\end{equation}
The fundamental character ratio is given by the ratio of modified Bessel functions:
\begin{equation}
\label{eq:zeta_half_su2}
\zeta_{1/2}(\beta) = \frac{c_{1/2}(\beta)}{c_0(\beta)} = \frac{I_2(\beta)}{I_1(\beta)}.
\end{equation}
For the adjoint representation $j = 1$ ($d_1 = 3$):
\begin{equation}
\label{eq:zeta_adj_su2}
\zeta_1(\beta) = \frac{I_3(\beta)}{I_1(\beta)}.
\end{equation}
\end{proposition}

\begin{proof}
Recall the Fourier-generating identity for modified Bessel functions of integer order:
\begin{equation}
e^{\beta \cos\theta} = I_0(\beta) + 2 \sum_{n=1}^\infty I_n(\beta) \cos(n\theta).
\end{equation}
Writing $\sin^2\theta = \frac{1}{2}(1 - \cos(2\theta))$, the partition function evaluates as:
\begin{align}
c_0(\beta) &= \frac{2}{\pi} \int_0^\pi d\theta \left( \frac{1 - \cos(2\theta)}{2} \right) \left[ I_0(\beta) + 2 \sum_{n=1}^\infty I_n(\beta) \cos(n\theta) \right] \nonumber \\
&= I_0(\beta) - I_2(\beta).
\end{align}
Using the standard Bessel recurrence relation:
\begin{equation}
\label{eq:bessel_recurrence_relation}
I_{n-1}(z) - I_{n+1}(z) = \frac{2n}{z} I_n(z)
\end{equation}
with $n=1$ yields:
\begin{equation}
I_0(\beta) - I_2(\beta) = \frac{2}{\beta} I_1(\beta).
\end{equation}
This establishes \eqref{eq:c0_su2}.

For general spin $j$, we use the product-to-sum trigonometric identity:
\begin{equation}
\sin\theta \sin((2j+1)\theta) = \frac{1}{2} \left[ \cos(2j\theta) - \cos((2j+2)\theta) \right].
\end{equation}
Integrating against the cosine series for $e^{\beta \cos\theta}$ yields:
\begin{equation}
\frac{2}{\pi} \int_0^\pi d\theta \, \sin\theta \sin((2j+1)\theta) \, e^{\beta \cos\theta} = I_{2j}(\beta) - I_{2j+2}(\beta).
\end{equation}
Applying the recurrence relation \eqref{eq:bessel_recurrence_relation} at index $n = 2j+1$:
\begin{equation}
I_{2j}(\beta) - I_{2j+2}(\beta) = \frac{2(2j+1)}{\beta} I_{2j+1}(\beta).
\end{equation}
Dividing by $d_j = 2j+1$ gives \eqref{eq:cj_su2}. For $j = 1/2$, $d_{1/2} = 2$ and $2j+1 = 2$, so:
\begin{equation}
c_{1/2}(\beta) = \frac{1}{2} \left( \frac{4}{\beta} I_2(\beta) \right) = \frac{2}{\beta} I_2(\beta).
\end{equation}
Dividing by $c_0(\beta) = \frac{2}{\beta} I_1(\beta)$ proves:
\begin{equation}
\zeta_{1/2}(\beta) = \frac{I_2(\beta)}{I_1(\beta)}. \qedhere
\end{equation}
\end{proof}

\subsection{Sharp Analytical Enclosures via Amos--Tur\'an Bounds}
\label{subsec:amos_turan_bounds}

To establish rigorous mathematical certificates on $\zeta_{1/2}(\beta)$ without numerical approximation errors, we invoke the Amos--Tur\'an inequalities for ratios of modified Bessel functions \cite{Amos1974}.

\begin{theorem}[Sharp Enclosures for $\zeta_{1/2}(\beta)$]
\label{thm:amos_bessel_bounds}
For all $\beta > 0$, the fundamental character ratio $\zeta_{1/2}(\beta) = I_2(\beta)/I_1(\beta)$ satisfies the strictly rational analytical bounds:
\begin{equation}
\label{eq:amos_bessel_inequality}
\frac{\beta}{2 + \sqrt{4 + \beta^2}} < \zeta_{1/2}(\beta) < \frac{\beta}{\frac{3}{2} + \sqrt{\frac{9}{4} + \beta^2}}.
\end{equation}
\end{theorem}

\begin{proof}
By Amos \cite{Amos1974}, the ratio $r_\nu(x) = I_\nu(x) / I_{\nu-1}(x)$ for $\nu > 0$ and $x > 0$ satisfies:
\begin{equation}
\frac{x}{\nu + \sqrt{\nu^2 + x^2}} < r_\nu(x) < \frac{x}{\nu - \frac{1}{2} + \sqrt{\left(\nu - \frac{1}{2}\right)^2 + x^2}}.
\end{equation}
Setting $\nu = 2$ and $x = \beta$ immediately yields \eqref{eq:amos_bessel_inequality}.
\end{proof}

Evaluating the Amos bounds \eqref{eq:amos_bessel_inequality} across the marginal crossover window $\mathcal{W} = [1/2, 5/2]$:
\begin{enumerate}
    \item At the infrared crossover boundary $\beta = 1/2$:
    \begin{equation}
    0.1231 < \zeta_{1/2}(1/2) < 0.1623.
    \end{equation}
    \item At the midpoint $\beta = 3/2$:
    \begin{equation}
    0.3333 < \zeta_{1/2}(3/2) < 0.4142.
    \end{equation}
    \item At the ultraviolet crossover boundary $\beta = 5/2$:
    \begin{equation}
    0.4806 < \zeta_{1/2}(5/2) < 0.5652.
    \end{equation}
\end{enumerate}
In particular, for all $\beta \le 2.0$, $\zeta_{1/2}(\beta) \le 0.4952 < 1/2$, proving strict contraction.

\section{Explicit Analytical Evaluation for \texorpdfstring{$\SU(3)$}{SU(3)}}
\label{sec:explicit_su3}

\subsection{Weyl Chamber Integration and Dynkin Labeling}
\label{subsec:su3_weyl_dynkin}

For $G = \SU(3)$, any element $U \in \SU(3)$ can be diagonalized into:
\begin{equation}
U \sim \mathrm{diag}\left( e^{i\theta_1}, e^{i\theta_2}, e^{-i(\theta_1 + \theta_2)} \right), \qquad \theta_1, \theta_2 \in [-\pi, \pi].
\end{equation}
Setting $\theta_3 = -(\theta_1 + \theta_2)$, the Haar measure on the Weyl chamber is:
\begin{equation}
\label{eq:su3_reduced_haar}
dU = \frac{1}{3! \cdot 4\pi^2} \prod_{1 \le j < k \le 3} \left| e^{i\theta_j} - e^{i\theta_k} \right|^2 d\theta_1 d\theta_2 = \frac{2}{3\pi^2} \sin^2\left(\frac{\theta_1 - \theta_2}{2}\right) \sin^2\left(\frac{2\theta_1 + \theta_2}{2}\right) \sin^2\left(\frac{\theta_1 + 2\theta_2}{2}\right) d\theta_1 d\theta_2.
\end{equation}
Irreducible representations of $\SU(3)$ are classified by two Dynkin integers $(p, q) \in \mathbb{N}_0^2$. The representation dimension is given by the Weyl dimension formula:
\begin{equation}
\label{eq:weyl_dimension_su3}
d_{(p, q)} = \frac{1}{2} (p + 1)(q + 1)(p + q + 2).
\end{equation}
The low-lying representations are:
\begin{itemize}
    \item Trivial representation: $(0, 0)$, dimension $d_{(0,0)} = 1$.
    \item Fundamental representation: $\mathbf{3} = (1, 0)$, dimension $d_{(1,0)} = 3$.
    \item Complex conjugate fundamental: $\bar{\mathbf{3}} = (0, 1)$, dimension $d_{(0,1)} = 3$.
    \item Adjoint representation: $\mathbf{8} = (1, 1)$, dimension $d_{(1,1)} = 8$.
    \item Symmetric sextet: $\mathbf{6} = (2, 0)$, dimension $d_{(2,0)} = 6$.
    \item Decuplet: $\mathbf{10} = (3, 0)$, dimension $d_{(3,0)} = 10$.
\end{itemize}

\subsection{Fundamental Character Ratio and Asymptotic Expansions}
\label{subsec:su3_character_ratio_props}

The single-plaquette Boltzmann weight for $\SU(3)$ is:
\begin{equation}
w_P(U) = \exp\left( \frac{\beta}{3} \RePart \Tr(U) \right) = \exp\left( \frac{\beta}{3} \left( \cos\theta_1 + \cos\theta_2 + \cos(\theta_1 + \theta_2) \right) \right).
\end{equation}
The single-plaquette partition function is:
\begin{equation}
\label{eq:su3_c0_integral}
c_0(\beta) = \int_{\SU(3)} dU \exp\left( \frac{\beta}{3} \RePart \Tr(U) \right).
\end{equation}

\begin{proposition}[Single-Plaquette Ratios for $\SU(3)$]
\label{prop:su3_character_ratios}
The fundamental character ratio $\zeta_{\mathbf{3}}(\beta)$ for $\SU(3)$ is:
\begin{equation}
\label{eq:zeta_3_def}
\zeta_{\mathbf{3}}(\beta) \coloneqq \frac{c_{(1,0)}(\beta)}{c_{(0,0)}(\beta)} = \frac{1}{3 c_0(\beta)} \int_{\SU(3)} dU \, \exp\left( \frac{\beta}{3} \RePart \Tr(U) \right) \Tr(U^\dagger).
\end{equation}
The function $\zeta_{\mathbf{3}}(\beta)$ is strictly monotonically increasing on $[0, \infty)$, with asymptotic limits:
\begin{align}
\zeta_{\mathbf{3}}(\beta) &= \frac{\beta}{18} \left( 1 - \frac{\beta^2}{324} + \mathcal{O}(\beta^4) \right) \qquad \text{as } \beta \to 0, \label{eq:su3_strong_asymptotics} \\
\zeta_{\mathbf{3}}(\beta) &= 1 - \frac{8}{3\beta} + \mathcal{O}\left(\frac{1}{\beta^2}\right) \qquad \text{as } \beta \to \infty. \label{eq:su3_weak_asymptotics}
\end{align}
Throughout the crossover interval $\beta \in [1/2, 5/2]$:
\begin{equation}
\label{eq:su3_crossover_bounds}
0.0278 \le \zeta_{\mathbf{3}}(\beta) \le 0.3814 < 0.40.
\end{equation}
\end{proposition}

\begin{proof}
Strict monotonicity follows from the positivity of the variance:
\begin{equation}
\frac{d}{d\beta} \zeta_{\mathbf{3}}(\beta) = \frac{1}{9} \left[ \langle |\Tr(U)|^2 \rangle - |\langle \Tr(U) \rangle|^2 \right] = \frac{1}{9} \mathrm{Var}_{\mu_\beta}(\Tr(U)) > 0.
\end{equation}
The strong-coupling expansion is computed by expanding the exponential to third order:
\begin{equation}
\exp\left( \frac{\beta}{6} (\Tr U + \Tr U^\dagger) \right) = 1 + \frac{\beta}{6}(\Tr U + \Tr U^\dagger) + \frac{\beta^2}{72}(\Tr U + \Tr U^\dagger)^2 + \dots
\end{equation}
Integrating against $\Tr U^\dagger$ using the Haar integrals:
\begin{equation}
\int dU \, \Tr(U) \Tr(U^\dagger) = 1, \qquad \int dU \, (\Tr U)^3 = 1,
\end{equation}
yields the leading coefficient $c_{(1,0)}(\beta) = \frac{\beta}{18} + \mathcal{O}(\beta^2)$.
The weak-coupling expansion is obtained from the quadratic saddle-point expansion around $U = \mathbf{1}$, where the 8 gluonic fluctuations each contribute $1/(2\beta)$, yielding the coefficient $8/(3\beta)$ \cite{Creutz1983}. The numerical enclosure across $\beta \in [1/2, 5/2]$ is certified by the interval arithmetic evaluated in Appendix~\ref{app:interval_certificates}.
\end{proof}

This completes the group-theoretic character expansions for $\SU(N)$.
